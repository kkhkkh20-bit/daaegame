            name=R13_NAMES[person]
            layout=r13_side_name_layout(name)
            width=renpy.render(Text(name,font=R13_UI_BOLD_FONT,size=layout['size']),1280,160,0,0).get_size()[0]
            if width>layout['width']-92 or 78+layout['width']>488:
                raise Exception('이름표 넘침: '+person)
        return True
