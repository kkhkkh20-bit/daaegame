    def qa_gaze116_export():

            for height in (560,760,840,1302):
                profile=r13_dialogue_source_profile(person,'neutral',height)
                if not profile:continue
                for slot,frame in profile['frames'].items():
                    result.append(dict(person=person,height=height,slot=slot,canvas=list(profile['canvas']),frame=dict(frame),loadable=renpy.loadable(frame['file'])))
        with open('/tmp/gaze116/MODE-runtime.json','w') as f:json.dump(result,f,ensure_ascii=False,indent=2)

label qa_gaze116_gallery_label:
    $ renpy.set_physical_size((1280,720))
    $ qa_gaze116_export()
    $ r13_edit(location='school')
    call r13_scene(R13_PLACES['school'][1],R13_PLACES['school'][0],'calm')
    $ qa_gaze116_index=-1
'''.replace('MODE',mode)
for i,(p,e) in enumerate(rows):
 script+=f"    $ qa_gaze116_index={i}\n    $ qa_gaze116_rows=[({p!r},'시선과 표정 확인 {i+1:03d}',dict(expression={e!r},pose={e!r}))]\n    call r13_lines(qa_gaze116_rows,'qa:gaze116')\n"
script+='    $ qa_gaze116_index=999\n    jump r13_field\n\ntestcase qa_gaze116_gallery:\n    pause 1.0\n    click "처음부터"\n    click pos (900,600) until screen "r13_hub" timeout 120\n    run Jump("qa_gaze116_gallery_label")\n'

script+='    pause until eval (qa_gaze116_index==999) timeout 25\n    assert eval (all(row["loadable"] for row in __import__("json").load(open("/tmp/gaze116/'+mode+'-runtime.json"))))\n'
